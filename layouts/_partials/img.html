{{- /* Выводит из ресурса: растровые картинки уменьшаются до width и переводятся в WebP. */}} {{- $res := .res }} {{- $width := .width | default 1440 }} {{- if in (slice "jpg" "jpeg" "png" "webp" "tiff" "bmp") (strings.TrimPrefix "image/" $res.MediaType.SubType) }} {{- if gt $res.Width $width }} {{- $res = $res.Resize (printf "%dx webp" $width) }} {{- else }} {{- $res = $res.Process "webp" }} {{- end }} {{- else }} {{- end }}
